emu-menu.c:

c
#include <SDL2/SDL.h>
#include <SDL2/SDL_ttf.h>
#include <dirent.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

#define MAX_ITEMS 256
#define MAX_PATH  512

typedef struct {
    const char *label;
    const char *core_path;
    const char *rom_dir;
} SystemEntry;

static SystemEntry systems[] = {
    { "SNES (snes9x2010)", "/usr/lib/libretro/snes9x2010_libretro.so", "/roms/snes" },
    { "NES (fceumm)",      "/usr/lib/libretro/fceumm_libretro.so",     "/roms/nes" },
    { "Genesis (picodrive)", "/usr/lib/libretro/picodrive_libretro.so","/roms/genesis" },
    { "Arcade (mame2003-plus)", "/usr/lib/libretro/mame2003_plus_libretro.so", "/roms/arcade" },
    { "PS1 (pcsx_rearmed)", "/usr/lib/libretro/pcsx_rearmed_libretro.so", "/roms/ps1" },
};
static const int system_count = sizeof(systems) / sizeof(systems[0]);

typedef struct {
    char name[256];
    char path[MAX_PATH];
} RomEntry;

static int list_roms(const char *dir, RomEntry *out, int max)
{
    DIR *d = opendir(dir);
    if (!d) return 0;

    struct dirent *de;
    int count = 0;
    while ((de = readdir(d)) != NULL && count < max) {
        if (de->d_name[0] == '.') continue;
        snprintf(out[count].name, sizeof(out[count].name), "%s", de->d_name);
        snprintf(out[count].path, sizeof(out[count].path), "%s/%s", dir, de->d_name);
        count++;
    }
    closedir(d);
    return count;
}

static void launch_retroarch(const char *core, const char *rom)
{
    const char *argv[] = {
        "/usr/bin/retroarch",
        "-L", core,
        rom,
        NULL
    };
    execvp(argv[0], (char * const *)argv);
    perror("execvp");
    _exit(1);
}

static void draw_menu(SDL_Renderer *ren, TTF_Font *font, const char *title,
                      char items[][256], int count, int selected)
{
    SDL_SetRenderDrawColor(ren, 0, 0, 0, 255);
    SDL_RenderClear(ren);

    int w, h;
    SDL_GetRendererOutputSize(ren, &w, &h);

    SDL_Color white = { 255, 255, 255, 255 };
    SDL_Color yellow = { 255, 255, 0, 255 };

    // Title
    if (title && font) {
        SDL_Surface *ts = TTF_RenderUTF8_Solid(font, title, white);
        if (ts) {
            SDL_Texture *tt = SDL_CreateTextureFromSurface(ren, ts);
            SDL_Rect dst = { 20, 20, ts->w, ts->h };
            SDL_RenderCopy(ren, tt, NULL, &dst);
            SDL_DestroyTexture(tt);
            SDL_FreeSurface(ts);
        }
    }

    int y = 80;
    for (int i = 0; i < count; i++) {
        SDL_Color col = (i == selected) ? yellow : white;
        SDL_Surface *s = TTF_RenderUTF8_Solid(font, items[i], col);
        if (!s) continue;
        SDL_Texture *t = SDL_CreateTextureFromSurface(ren, s);
        SDL_Rect dst = { 40, y, s->w, s->h };
        SDL_RenderCopy(ren, t, NULL, &dst);
        SDL_DestroyTexture(t);
        SDL_FreeSurface(s);
        y += 30;
    }

    SDL_RenderPresent(ren);
}

static int menu_loop(SDL_Renderer *ren, TTF_Font *font,
                     const char *title, char items[][256], int count)
{
    int selected = 0;
    int running = 1;

    SDL_GameControllerEventState(SDL_ENABLE);
    for (int i = 0; i < SDL_NumJoysticks(); i++) {
        if (SDL_IsGameController(i)) {
            SDL_GameControllerOpen(i);
        }
    }

    while (running) {
        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            switch (e.type) {
            case SDL_QUIT:
                return -1;
            case SDL_KEYDOWN:
                if (e.key.keysym.sym == SDLK_ESCAPE) return -1;
                if (e.key.keysym.sym == SDLK_UP) {
                    if (selected > 0) selected--;
                } else if (e.key.keysym.sym == SDLK_DOWN) {
                    if (selected < count - 1) selected++;
                } else if (e.key.keysym.sym == SDLK_RETURN ||
                           e.key.keysym.sym == SDLK_SPACE) {
                    return selected;
                }
                break;
            case SDL_CONTROLLERBUTTONDOWN:
                if (e.cbutton.button == SDL_CONTROLLER_BUTTON_DPAD_UP ||
                    e.cbutton.button == SDL_CONTROLLER_BUTTON_LEFTSTICK_UP) {
                    if (selected > 0) selected--;
                } else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_DPAD_DOWN ||
                           e.cbutton.button == SDL_CONTROLLER_BUTTON_LEFTSTICK_DOWN) {
                    if (selected < count - 1) selected++;
                } else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_A) {
                    return selected;
                } else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_B) {
                    return -1;
                }
                break;
            default:
                break;
            }
        }

        draw_menu(ren, font, title, items, count, selected);
        SDL_Delay(16);
    }
    return -1;
}

int main(int argc, char **argv)
{
    if (SDL_Init(SDL_INIT_VIDEO | SDL_INIT_GAMECONTROLLER | SDL_INIT_AUDIO) != 0) {
        fprintf(stderr, "SDL_Init: %s\n", SDL_GetError());
        return 1;
    }

    if (TTF_Init() != 0) {
        fprintf(stderr, "TTF_Init: %s\n", TTF_GetError());
        return 1;
    }

    SDL_Window *win = SDL_CreateWindow("EMU Menu",
                                       SDL_WINDOWPOS_CENTERED, SDL_WINDOWPOS_CENTERED,
                                       800, 600,
                                       SDL_WINDOW_FULLSCREEN_DESKTOP);
    if (!win) {
        fprintf(stderr, "SDL_CreateWindow: %s\n", SDL_GetError());
        return 1;
    }

    SDL_Renderer *ren = SDL_CreateRenderer(win, -1,
                                           SDL_RENDERER_ACCELERATED |
                                           SDL_RENDERER_PRESENTVSYNC);
    if (!ren) {
        fprintf(stderr, "SDL_CreateRenderer: %s\n", SDL_GetError());
        return 1;
    }

    TTF_Font *font = TTF_OpenFont("/usr/share/fonts/ttf-dejavu/DejaVuSans.ttf", 20);
    if (!font) {
        fprintf(stderr, "TTF_OpenFont: %s\n", TTF_GetError());
        return 1;
    }

    // Build system list
    char sys_items[MAX_ITEMS][256];
    for (int i = 0; i < system_count; i++) {
        snprintf(sys_items[i], sizeof(sys_items[i]), "%s", systems[i].label);
    }

    while (1) {
        int sys_idx = menu_loop(ren, font, "Select System", sys_items, system_count);
        if (sys_idx < 0) break;

        SystemEntry *sys = &systems[sys_idx];

        RomEntry roms[MAX_ITEMS];
        int rom_count = list_roms(sys->rom_dir, roms, MAX_ITEMS);
        if (rom_count <= 0) {
            // No ROMs, go back
            continue;
        }

        char rom_items[MAX_ITEMS][256];
        for (int i = 0; i < rom_count; i++) {
            snprintf(rom_items[i], sizeof(rom_items[i]), "%s", roms[i].name);
        }

        int rom_idx = menu_loop(ren, font, sys->label, rom_items, rom_count);
        if (rom_idx < 0) continue;

        // Launch RetroArch with selected core + ROM
        launch_retroarch(sys->core_path, roms[rom_idx].path);
        // If exec fails, we’ll just exit
        break;
    }

    TTF_CloseFont(font);
    SDL_DestroyRenderer(ren);
    SDL_DestroyWindow(win);
    TTF_Quit();
    SDL_Quit();
    return 0;
}

static void draw_menu(SDL_Renderer *ren, TTF_Font *font, const char *title,
                      char items[][256], int count, int selected, Uint8 alpha)
{
    int w, h;
    SDL_GetRendererOutputSize(ren, &w, &h);

    // Clear with transparent black (we assume mpv is behind)
    SDL_SetRenderDrawBlendMode(ren, SDL_BLENDMODE_BLEND);
    SDL_SetRenderDrawColor(ren, 0, 0, 0, 0);
    SDL_RenderClear(ren);

    // Translucent panel
    SDL_SetRenderDrawColor(ren, 0, 0, 0, alpha);
    SDL_Rect panel = { w / 8, h / 8, w * 3 / 4, h * 3 / 4 };
    SDL_RenderFillRect(ren, &panel);

    SDL_Color white = { 255, 255, 255, 255 };
    SDL_Color yellow = { 255, 255, 0, 255 };

    int y = panel.y + 20;

    if (title && font) {
        SDL_Surface *ts = TTF_RenderUTF8_Solid(font, title, white);
        if (ts) {
            SDL_Texture *tt = SDL_CreateTextureFromSurface(ren, ts);
            SDL_Rect dst = { panel.x + 20, y, ts->w, ts->h };
            SDL_RenderCopy(ren, tt, NULL, &dst);
            SDL_DestroyTexture(tt);
            SDL_FreeSurface(ts);
            y += ts->h + 20;
        }
    }

    for (int i = 0; i < count; i++) {
        SDL_Color col = (i == selected) ? yellow : white;
        SDL_Surface *s = TTF_RenderUTF8_Solid(font, items[i], col);
        if (!s) continue;
        SDL_Texture *t = SDL_CreateTextureFromSurface(ren, s);
        SDL_Rect dst = { panel.x + 40, y, s->w, s->h };
        SDL_RenderCopy(ren, t, NULL, &dst);
        SDL_DestroyTexture(t);
        SDL_FreeSurface(s);
        y += 30;
    }

    SDL_RenderPresent(ren);
}
Update menu_loop to fade in:

c
static int menu_loop(SDL_Renderer *ren, TTF_Font *font,
                     const char *title, char items[][256], int count)
{
    int selected = 0;
    int running = 1;
    Uint8 alpha = 0;

    SDL_GameControllerEventState(SDL_ENABLE);
    for (int i = 0; i < SDL_NumJoysticks(); i++) {
        if (SDL_IsGameController(i)) {
            SDL_GameControllerOpen(i);
        }
    }

    // Fade-in
    while (alpha < 180) {
        draw_menu(ren, font, title, items, count, selected, alpha);
        alpha += 10;
        SDL_Delay(16);
    }
    alpha = 180;

    while (running) {
        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            switch (e.type) {
            case SDL_QUIT:
                return -1;
            case SDL_KEYDOWN:
                if (e.key.keysym.sym == SDLK_ESCAPE) return -1;
                if (e.key.keysym.sym == SDLK_UP && selected > 0) selected--;
                else if (e.key.keysym.sym == SDLK_DOWN && selected < count - 1) selected++;
                else if (e.key.keysym.sym == SDLK_RETURN || e.key.keysym.sym == SDLK_SPACE)
                    return selected;
                break;
            case SDL_CONTROLLERBUTTONDOWN:
                if (e.cbutton.button == SDL_CONTROLLER_BUTTON_DPAD_UP && selected > 0)
                    selected--;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_DPAD_DOWN && selected < count - 1)
                    selected++;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_A)
                    return selected;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_B)
                    return -1;
                break;
            default:
                break;
            }
        }

        draw_menu(ren, font, title, items, count, selected, alpha);
        SDL_Delay(16);
    }
    return -1;
}

#include <signal.h>

static void mpv_pause(void)
{
    const char *pid_str = getenv("MPV_PID");
    if (!pid_str) return;
    pid_t pid = (pid_t)atoi(pid_str);
    if (pid > 1) kill(pid, SIGSTOP);
}

static void mpv_resume(void)
{
    const char *pid_str = getenv("MPV_PID");
    if (!pid_str) return;
    pid_t pid = (pid_t)atoi(pid_str);
    if (pid > 1) kill(pid, SIGCONT);
}
Update launch_retroarch:

c
static void launch_retroarch(const char *core, const char *rom)
{
    mpv_pause();

    const char *argv[] = {
        "/usr/bin/retroarch",
        "-L", core,
        rom,
        NULL
    };
    execvp(argv[0], (char * const *)argv);
    perror("execvp");

    mpv_resume(); // only if exec fails
    _exit(1);
}

4.2. Settings data + config I/O
c
#define CFG_PATH "/config/emu.conf"

typedef struct {
    int volume;        // 0-100
    int bg_index;      // index into bg_keys[]
} Settings;

static const char *bg_keys[] = {
    "global",
    "snes",
    "nes",
    "genesis",
    "arcade",
    "ps1",
};
static const int bg_key_count = sizeof(bg_keys)/sizeof(bg_keys[0]);

static void load_settings(Settings *s)
{
    s->volume = 60;
    s->bg_index = 0;

    FILE *f = fopen(CFG_PATH, "r");
    if (!f) return;

    char line[128];
    while (fgets(line, sizeof(line), f)) {
        if (sscanf(line, "VOLUME=%d", &s->volume) == 1) continue;
        if (sscanf(line, "BG_INDEX=%d", &s->bg_index) == 1) continue;
    }
    fclose(f);

    if (s->volume < 0) s->volume = 0;
    if (s->volume > 100) s->volume = 100;
    if (s->bg_index < 0 || s->bg_index >= bg_key_count) s->bg_index = 0;
}

static void save_settings(const Settings *s)
{
    FILE *f = fopen(CFG_PATH, "w");
    if (!f) return;
    fprintf(f, "VOLUME=%d\n", s->volume);
    fprintf(f, "BG_INDEX=%d\n", s->bg_index);
    fclose(f);
}
4.3. Settings menu
c
static void run_settings_menu(SDL_Renderer *ren, TTF_Font *font)
{
    Settings s;
    load_settings(&s);

    int selected = 0; // 0 = volume, 1 = background, 2 = back
    int running = 1;
    Uint8 alpha = 180;

    while (running) {
        char items[3][256];
        snprintf(items[0], sizeof(items[0]), "Volume: %d", s.volume);
        snprintf(items[1], sizeof(items[1]), "Background: %s", bg_keys[s.bg_index]);
        snprintf(items[2], sizeof(items[2]), "Back");

        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            if (e.type == SDL_QUIT) return;
            if (e.type == SDL_KEYDOWN) {
                switch (e.key.keysym.sym) {
                case SDLK_ESCAPE:
                    running = 0;
                    break;
                case SDLK_UP:
                    if (selected > 0) selected--;
                    break;
                case SDLK_DOWN:
                    if (selected < 2) selected++;
                    break;
                case SDLK_LEFT:
                    if (selected == 0 && s.volume > 0) s.volume -= 5;
                    if (selected == 1 && s.bg_index > 0) s.bg_index--;
                    break;
                case SDLK_RIGHT:
                    if (selected == 0 && s.volume < 100) s.volume += 5;
                    if (selected == 1 && s.bg_index < bg_key_count - 1) s.bg_index++;
                    break;
                case SDLK_RETURN:
                case SDLK_SPACE:
                    if (selected == 2) running = 0;
                    break;
                default:
                    break;
                }
            }
        }

        draw_menu(ren, font, "Settings", items, 3, selected, alpha);
        SDL_Delay(16);
    }

    save_settings(&s);

    // Apply new global background immediately
    char cmd[128];
    snprintf(cmd, sizeof(cmd), "emu-bg %s &", bg_keys[s.bg_index]);
    system(cmd);
}

#include <SDL2/SDL.h>
#include <SDL2/SDL_ttf.h>
#include <SDL2/SDL_image.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <unistd.h>

#define MAX_ITEMS 256
#define CFG_PATH "/config/emu.conf"

typedef struct {
    const char *label;
    const char *core_path;
    const char *rom_dir;
    const char *bg_key;
} SystemEntry;

static SystemEntry systems[] = {
    { "SNES (snes9x2010)", "/usr/lib/libretro/snes9x2010_libretro.so", "/roms/snes",   "snes" },
    { "NES (fceumm)",      "/usr/lib/libretro/fceumm_libretro.so",     "/roms/nes",    "nes" },
    { "Genesis (picodrive)", "/usr/lib/libretro/picodrive_libretro.so","/roms/genesis","genesis" },
    { "Arcade (mame2003-plus)", "/usr/lib/libretro/mame2003_plus_libretro.so", "/roms/arcade", "arcade" },
    { "PS1 (pcsx_rearmed)", "/usr/lib/libretro/pcsx_rearmed_libretro.so", "/roms/ps1", "ps1" },
};
static const int system_count = sizeof(systems)/sizeof(systems[0]);

typedef struct {
    int volume;   // 0-100
    int bg_index; // index into bg_keys[]
} Settings;

static const char *bg_keys[] = {
    "global",
    "snes",
    "nes",
    "genesis",
    "arcade",
    "ps1",
};
static const int bg_key_count = sizeof(bg_keys)/sizeof(bg_keys[0]);

static void draw_menu(SDL_Renderer *ren, TTF_Font *font, const char *title,
                      char items[][256], int count, int selected, Uint8 alpha)
{
    int w, h;
    SDL_GetRendererOutputSize(ren, &w, &h);

    SDL_SetRenderDrawBlendMode(ren, SDL_BLENDMODE_BLEND);
    SDL_SetRenderDrawColor(ren, 0, 0, 0, 0);
    SDL_RenderClear(ren);

    SDL_SetRenderDrawColor(ren, 0, 0, 0, alpha);
    SDL_Rect panel = { w / 8, h / 8, w * 3 / 4, h * 3 / 4 };
    SDL_RenderFillRect(ren, &panel);

    SDL_Color white = { 255, 255, 255, 255 };
    SDL_Color yellow = { 255, 255, 0, 255 };

    int y = panel.y + 20;

    if (title && font) {
        SDL_Surface *ts = TTF_RenderUTF8_Solid(font, title, white);
        if (ts) {
            SDL_Texture *tt = SDL_CreateTextureFromSurface(ren, ts);
            SDL_Rect dst = { panel.x + 20, y, ts->w, ts->h };
            SDL_RenderCopy(ren, tt, NULL, &dst);
            SDL_DestroyTexture(tt);
            y += ts->h + 20;
            SDL_FreeSurface(ts);
        }
    }

    for (int i = 0; i < count; i++) {
        SDL_Color col = (i == selected) ? yellow : white;
        SDL_Surface *s = TTF_RenderUTF8_Solid(font, items[i], col);
        if (!s) continue;
        SDL_Texture *t = SDL_CreateTextureFromSurface(ren, s);
        SDL_Rect dst = { panel.x + 40, y, s->w, s->h };
        SDL_RenderCopy(ren, t, NULL, &dst);
        SDL_DestroyTexture(t);
        SDL_FreeSurface(s);
        y += 30;
    }

    SDL_RenderPresent(ren);
}

static void load_settings(Settings *s)
{
    s->volume = 60;
    s->bg_index = 0;

    FILE *f = fopen(CFG_PATH, "r");
    if (!f) return;

    char line[128];
    while (fgets(line, sizeof(line), f)) {
        if (sscanf(line, "VOLUME=%d", &s->volume) == 1) continue;
        if (sscanf(line, "BG_INDEX=%d", &s->bg_index) == 1) continue;
    }
    fclose(f);

    if (s->volume < 0) s->volume = 0;
    if (s->volume > 100) s->volume = 100;
    if (s->bg_index < 0 || s->bg_index >= bg_key_count) s->bg_index = 0;
}

static void save_settings(const Settings *s)
{
    FILE *f = fopen(CFG_PATH, "w");
    if (!f) return;
    fprintf(f, "VOLUME=%d\n", s->volume);
    fprintf(f, "BG_INDEX=%d\n", s->bg_index);
    fclose(f);
}

static void run_settings_menu(SDL_Renderer *ren, TTF_Font *font)
{
    Settings s;
    load_settings(&s);

    int selected = 0; // 0 = volume, 1 = background, 2 = back
    int running = 1;
    Uint8 alpha = 180;

    while (running) {
        char items[3][256];
        snprintf(items[0], sizeof(items[0]), "Volume: %d", s.volume);
        snprintf(items[1], sizeof(items[1]), "Background: %s", bg_keys[s.bg_index]);
        snprintf(items[2], sizeof(items[2]), "Back");

        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            if (e.type == SDL_QUIT) return;
            if (e.type == SDL_KEYDOWN) {
                switch (e.key.keysym.sym) {
                case SDLK_ESCAPE:
                    running = 0;
                    break;
                case SDLK_UP:
                    if (selected > 0) selected--;
                    break;
                case SDLK_DOWN:
                    if (selected < 2) selected++;
                    break;
                case SDLK_LEFT:
                    if (selected == 0 && s.volume > 0) s.volume -= 5;
                    if (selected == 1 && s.bg_index > 0) s.bg_index--;
                    break;
                case SDLK_RIGHT:
                    if (selected == 0 && s.volume < 100) s.volume += 5;
                    if (selected == 1 && s.bg_index < bg_key_count - 1) s.bg_index++;
                    break;
                case SDLK_RETURN:
                case SDLK_SPACE:
                    if (selected == 2) running = 0;
                    break;
                default:
                    break;
                }
            }
        }

        draw_menu(ren, font, "Settings", items, 3, selected, alpha);
        SDL_Delay(16);
    }

    save_settings(&s);

    char cmd[128];
    snprintf(cmd, sizeof(cmd), "emu-bg %s &", bg_keys[s.bg_index]);
    system(cmd);
}

static void launch_retroarch(const char *core, const char *rom)
{
    const char *argv[] = {
        "/usr/bin/retroarch",
        "-L", core,
        rom,
        NULL
    };
    execvp(argv[0], (char * const *)argv);
    perror("execvp");
    _exit(1);
}

static int menu_loop(SDL_Renderer *ren, TTF_Font *font,
                     const char *title, char items[][256], int count)
{
    int selected = 0;
    int running = 1;
    Uint8 alpha = 0;

    SDL_GameControllerEventState(SDL_ENABLE);
    for (int i = 0; i < SDL_NumJoysticks(); i++) {
        if (SDL_IsGameController(i)) {
            SDL_GameControllerOpen(i);
        }
    }

    while (alpha < 180) {
        draw_menu(ren, font, title, items, count, selected, alpha);
        alpha += 10;
        SDL_Delay(16);
    }
    alpha = 180;

    while (running) {
        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            switch (e.type) {
            case SDL_QUIT:
                return -1;
            case SDL_KEYDOWN:
                if (e.key.keysym.sym == SDLK_ESCAPE) return -1;
                if (e.key.keysym.sym == SDLK_UP && selected > 0) selected--;
                else if (e.key.keysym.sym == SDLK_DOWN && selected < count - 1) selected++;
                else if (e.key.keysym.sym == SDLK_RETURN || e.key.keysym.sym == SDLK_SPACE)
                    return selected;
                break;
            case SDL_CONTROLLERBUTTONDOWN:
                if (e.cbutton.button == SDL_CONTROLLER_BUTTON_DPAD_UP && selected > 0)
                    selected--;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_DPAD_DOWN && selected < count - 1)
                    selected++;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_A)
                    return selected;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_B)
                    return -1;
                break;
            default:
                break;
            }
        }

        draw_menu(ren, font, title, items, count, selected, alpha);
        SDL_Delay(16);
    }
    return -1;
}

static int list_roms(const char *dir, char items[][256])
{
    DIR *d = opendir(dir);
    if (!d) return 0;
    struct dirent *ent;
    int n = 0;
    while ((ent = readdir(d)) && n < MAX_ITEMS) {
        if (ent->d_name[0] == '.') continue;
        snprintf(items[n], 256, "%s", ent->d_name);
        n++;
    }
    closedir(d);
    return n;
}

int main(int argc, char **argv)
{
    if (SDL_Init(SDL_INIT_VIDEO | SDL_INIT_GAMECONTROLLER) != 0) {
        fprintf(stderr, "SDL_Init: %s\n", SDL_GetError());
        return 1;
    }
    if (TTF_Init() != 0) {
        fprintf(stderr, "TTF_Init: %s\n", TTF_GetError());
        return 1;
    }

    SDL_Window *win = SDL_CreateWindow("emu-os",
                                       SDL_WINDOWPOS_UNDEFINED,
                                       SDL_WINDOWPOS_UNDEFINED,
                                       1280, 720,
                                       SDL_WINDOW_FULLSCREEN);
    if (!win) {
        fprintf(stderr, "SDL_CreateWindow: %s\n", SDL_GetError());
        return 1;
    }

    SDL_Renderer *ren = SDL_CreateRenderer(win, -1, SDL_RENDERER_ACCELERATED);
    if (!ren) {
        fprintf(stderr, "SDL_CreateRenderer: %s\n", SDL_GetError());
        return 1;
    }

    TTF_Font *font = TTF_OpenFont("/usr/share/fonts/ttf-dejavu/DejaVuSans.ttf", 24);
    if (!font) {
        fprintf(stderr, "TTF_OpenFont: %s\n", TTF_GetError());
        return 1;
    }

    int running = 1;
    while (running) {
        char sys_items[MAX_ITEMS][256];
        for (int i = 0; i < system_count; i++) {
            snprintf(sys_items[i], sizeof(sys_items[i]), "%s", systems[i].label);
        }
        snprintf(sys_items[system_count], sizeof(sys_items[system_count]), "Settings");
        int total = system_count + 1;

        int sys_idx = menu_loop(ren, font, "Select System", sys_items, total);
        if (sys_idx < 0) break;

        if (sys_idx == system_count) {
            run_settings_menu(ren, font);
            continue;
        }

        SystemEntry *sys = &systems[sys_idx];

        char cmd[128];
        snprintf(cmd, sizeof(cmd), "emu-bg %s &", sys->bg_key);
        system(cmd);

        char rom_items[MAX_ITEMS][256];
        int rom_count = list_roms(sys->rom_dir, rom_items);
        if (rom_count <= 0) continue;

        int rom_idx = menu_loop(ren, font, sys->label, rom_items, rom_count);
        if (rom_idx < 0) continue;

        char rom_path[512];
        snprintf(rom_path, sizeof(rom_path), "%s/%s", sys->rom_dir, rom_items[rom_idx]);

        launch_retroarch(sys->core_path, rom_path);
    }

    TTF_CloseFont(font);
    SDL_DestroyRenderer(ren);
    SDL_DestroyWindow(win);
    TTF_Quit();
    SDL_Quit();
    return 0;
}

🧱 4. Auto‑mount Script: overlay/etc/init.d/S20mount_roms
sh
#!/bin/sh

### BEGIN INIT INFO
# Provides:          mount_roms
# Required-Start:    udev
# Default-Start:     S
### END INIT INFO

ROM_MNT="/mnt/roms"
mkdir -p "$ROM_MNT"

# Detect second partition of boot device
DEV=""
for d in /dev/sd?? /dev/mmcblk?p?; do
    case "$d" in
        *2) DEV="$d" ;;
    esac
done

if [ -n "$DEV" ]; then
    echo "Mounting ROM partition: $DEV"
    mount -t exfat -o uid=0,gid=0,umask=000 "$DEV" "$ROM_MNT"
else
    echo "No ROM partition found"
fi

🧱 5. Linux USB Installer Script
This script:

✔ Partitions the USB
✔ Formats partition 1 as ext4
✔ Formats partition 2 as exFAT
✔ Installs Syslinux
✔ Copies kernel + initrd
✔ Creates syslinux.cfg
✔ Labels the partitions
✔ Makes the USB fully bootable
Save as: install_emu_os_usb.sh

bash
#!/usr/bin/env bash
set -e

if [ "$EUID" -ne 0 ]; then
    echo "Run as root"
    exit 1
fi

if [ $# -ne 3 ]; then
    echo "Usage: $0 <USB device> <bzImage> <rootfs.cpio.gz>"
    echo "Example: $0 /dev/sdb bzImage rootfs.cpio.gz"
    exit 1
fi

DEV="$1"
KERNEL="$2"
INITRD="$3"

echo "WARNING: This will erase ALL data on $DEV"
read -p "Continue? (yes/no): " yn
[ "$yn" = "yes" ] || exit 1

echo "Partitioning $DEV"
sgdisk --zap-all "$DEV"
sgdisk -n 1:0:+2G -t 1:8300 -c 1:"EMU-OS" "$DEV"
sgdisk -n 2:0:0   -t 2:0700 -c 2:"ROMDATA" "$DEV"

partprobe "$DEV"

P1="${DEV}1"
P2="${DEV}2"

echo "Formatting partitions"
mkfs.ext4 -L EMU-OS "$P1"
mkfs.exfat -n ROMDATA "$P2"

echo "Mounting boot partition"
mkdir -p /mnt/emuos
mount "$P1" /mnt/emuos

echo "Copying kernel and initrd"
cp "$KERNEL" /mnt/emuos/bzImage
cp "$INITRD" /mnt/emuos/rootfs.cpio.gz

echo "Installing Syslinux"
syslinux --install "$P1"
dd if=/usr/lib/syslinux/mbr/mbr.bin of="$DEV"

echo "Creating syslinux.cfg"
cat > /mnt/emuos/syslinux.cfg <<EOF
DEFAULT emu
LABEL emu
    KERNEL /bzImage
    APPEND initrd=/rootfs.cpio.gz quiet
EOF

sync
umount /mnt/emuos
rmdir /mnt/emuos

echo "USB installation complete"
echo "Partition 2 (ROMDATA) is ready for ROMs"

1. Wipes the USB drive

2. Creates two partitions

3. Formats them

whats a USB first boot setup wizard ?
A USB first‑boot setup wizard is a small program that runs the very first time emu‑os boots from the USB stick.
Its job is to automatically prepare the user’s ROM partition, settings, and environment so the system is ready to use without requiring any Linux knowledge.

Think of it as the “initial setup” screen you see on game consoles — but minimal, fast, and tailored to your emulator OS.

Here’s what it does and why it’s useful.

🎮 What a USB First‑Boot Setup Wizard Does
A first‑boot wizard typically performs one‑time initialization tasks such as:

1. Detect the ROM partition
Finds /dev/sdX2 (the exFAT partition)

Verifies it is mountable

Mounts it to /mnt/roms

If the partition is missing or corrupted, it can:

Recreate it

Reformat it

Or prompt the user to fix the USB stick

2. Create the ROM directory structure
If the user hasn’t created folders yet, the wizard does it automatically:

Code
/mnt/roms/snes
/mnt/roms/nes
/mnt/roms/genesis
/mnt/roms/arcade
/mnt/roms/ps1
This ensures the menu won’t show “empty” or “missing directory” errors.

3. Create default config files
For example:

/config/emu.conf

/config/retroarch.cfg (optional)

/config/controller-mapping.cfg

This prevents missing‑file crashes and gives the user a working baseline.

4. Set default background video
If the user hasn’t chosen one yet:

Code
BG_INDEX=0
Or it can auto‑select based on detected ROMs.

5. Run a quick hardware check
Optional but useful:

Detects audio output (HDMI vs analog)

Detects connected controllers

Detects GPU acceleration support

Detects display resolution

This allows the wizard to set sane defaults.

6. Show a simple “Welcome” screen
Something like:

Code
Welcome to emu‑os!
Your ROM storage has been prepared.
Drop your ROMs into the ROMDATA partition.
Press A to continue.
After this, the wizard never runs again unless the user deletes a flag file.

🧠 Why You Might Want This
A first‑boot wizard makes your OS feel:

More polished

More user‑friendly

More console‑like

More resilient to user error

It also prevents:

Missing ROM folders

Missing config files

Incorrect mount points

Broken audio/video defaults

And it makes the system “just work” even for non‑technical users.

🛠️ How It Works Internally
On first boot, the wizard runs because a flag file is missing:

Code
/config/.firstboot_done
After it completes, it creates the flag:

Code
touch /config/.firstboot_done
On future boots, the wizard sees the flag and exits instantly.

Create emu-setup-src/emu-setup.c (new directory, similar to emu-menu-src). Here’s a compact but complete skeleton:

c
#include <SDL2/SDL.h>
#include <SDL2/SDL_ttf.h>
#include <stdio.h>
#include <sys/stat.h>
#include <unistd.h>

#define FLAG_PATH "/config/.firstboot_done"

static void ensure_dir(const char *path) {
    mkdir(path, 0755);
}

static void prepare_rom_dirs(void) {
    ensure_dir("/mnt/roms");
    ensure_dir("/mnt/roms/snes");
    ensure_dir("/mnt/roms/nes");
    ensure_dir("/mnt/roms/genesis");
    ensure_dir("/mnt/roms/arcade");
    ensure_dir("/mnt/roms/ps1");
}

static void prepare_config_files(void) {
    ensure_dir("/config");
    FILE *f = fopen("/config/emu.conf", "w");
    if (f) {
        fprintf(f, "VOLUME=60\n");
        fprintf(f, "BG_INDEX=0\n");
        fclose(f);
    }
    // RetroArch config will be provided via overlay; this just ensures /config exists.
}

static void detect_controllers_and_audio(void) {
    SDL_InitSubSystem(SDL_INIT_GAMECONTROLLER | SDL_INIT_AUDIO);

    int pads = 0;
    for (int i = 0; i < SDL_NumJoysticks(); i++) {
        if (SDL_IsGameController(i)) pads++;
    }
    SDL_AudioSpec spec;
    if (SDL_GetAudioDeviceName(0, 0)) {
        // We could log or store this in a config if desired.
    }
    (void)pads;
}

int main(int argc, char **argv) {
    if (access(FLAG_PATH, F_OK) == 0) {
        return 0;
    }

    if (SDL_Init(SDL_INIT_VIDEO | SDL_INIT_GAMECONTROLLER | SDL_INIT_AUDIO) != 0) {
        return 1;
    }
    if (TTF_Init() != 0) {
        SDL_Quit();
        return 1;
    }

    SDL_Window *win = SDL_CreateWindow("emu-os setup",
                                       SDL_WINDOWPOS_UNDEFINED,
                                       SDL_WINDOWPOS_UNDEFINED,
                                       1280, 720,
                                       SDL_WINDOW_FULLSCREEN);
    SDL_Renderer *ren = SDL_CreateRenderer(win, -1, SDL_RENDERER_ACCELERATED);

    // Custom background: load a PNG/JPG from /usr/share/emu/setup_bg.png if you like.
    // For now, just a solid color + text.
    TTF_Font *font = TTF_OpenFont("/usr/share/fonts/ttf-dejavu/DejaVuSans.ttf", 28);

    prepare_rom_dirs();
    prepare_config_files();
    detect_controllers_and_audio();

    int done = 0;
    while (!done) {
        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            if (e.type == SDL_QUIT) done = 1;
            if (e.type == SDL_KEYDOWN) {
                if (e.key.keysym.sym == SDLK_RETURN || e.key.keysym.sym == SDLK_SPACE)
                    done = 1;
            }
            if (e.type == SDL_CONTROLLERBUTTONDOWN) {
                if (e.cbutton.button == SDL_CONTROLLER_BUTTON_A)
                    done = 1;
            }
        }

        SDL_SetRenderDrawColor(ren, 0, 0, 0, 255);
        SDL_RenderClear(ren);

        if (font) {
            SDL_Color white = {255, 255, 255, 255};
            SDL_Surface *s = TTF_RenderUTF8_Solid(font,
                "Welcome to emu-os! ROM storage and configs are prepared.",
                white);
            if (s) {
                SDL_Texture *t = SDL_CreateTextureFromSurface(ren, s);
                SDL_Rect dst = { 80, 300, s->w, s->h };
                SDL_RenderCopy(ren, t, NULL, &dst);
                SDL_DestroyTexture(t);
                SDL_FreeSurface(s);
            }
        }

        SDL_RenderPresent(ren);
        SDL_Delay(16);
    }

    FILE *flag = fopen(FLAG_PATH, "w");
    if (flag) fclose(flag);

    if (font) TTF_CloseFont(font);
    SDL_DestroyRenderer(ren);
    SDL_DestroyWindow(win);
    TTF_Quit();
    SDL_Quit();
    return 0;
}

cat > "$ROOT_DIR/emu-menu-src/emu-menu.c" << 'EOF'
#include <SDL2/SDL.h>
#include <SDL2/SDL_ttf.h>
#include <SDL2/SDL_image.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <unistd.h>

#define MAX_ITEMS 256
#define CFG_PATH "/config/emu.conf"

typedef struct {
    const char *label;
    const char *core_path;
    const char *rom_dir;
    const char *bg_key;
} SystemEntry;

static SystemEntry systems[] = {
    { "SNES (snes9x2010)", "/usr/lib/libretro/snes9x2010_libretro.so", "/mnt/roms/snes",   "snes" },
    { "NES (fceumm)",      "/usr/lib/libretro/fceumm_libretro.so",     "/mnt/roms/nes",    "nes" },
    { "Genesis (picodrive)", "/usr/lib/libretro/picodrive_libretro.so","/mnt/roms/genesis","genesis" },
    { "Arcade (mame2003-plus)", "/usr/lib/libretro/mame2003_plus_libretro.so", "/mnt/roms/arcade", "arcade" },
    { "PS1 (pcsx_rearmed)", "/usr/lib/libretro/pcsx_rearmed_libretro.so", "/mnt/roms/ps1", "ps1" },
};
static const int system_count = sizeof(systems)/sizeof(systems[0]);

typedef struct {
    int volume;   // 0-100
    int bg_index; // index into bg_keys[]
} Settings;

static const char *bg_keys[] = {
    "global",
    "snes",
    "nes",
    "genesis",
    "arcade",
    "ps1",
};
static const int bg_key_count = sizeof(bg_keys)/sizeof(bg_keys[0]);

static void draw_menu(SDL_Renderer *ren, TTF_Font *font, const char *title,
                      char items[][256], int count, int selected, Uint8 alpha)
{
    int w, h;
    SDL_GetRendererOutputSize(ren, &w, &h);

    SDL_SetRenderDrawBlendMode(ren, SDL_BLENDMODE_BLEND);
    SDL_SetRenderDrawColor(ren, 0, 0, 0, 0);
    SDL_RenderClear(ren);

    SDL_SetRenderDrawColor(ren, 0, 0, 0, alpha);
    SDL_Rect panel = { w / 8, h / 8, w * 3 / 4, h * 3 / 4 };
    SDL_RenderFillRect(ren, &panel);

    SDL_Color white = { 255, 255, 255, 255 };
    SDL_Color yellow = { 255, 255, 0, 255 };

    int y = panel.y + 20;

    if (title && font) {
        SDL_Surface *ts = TTF_RenderUTF8_Solid(font, title, white);
        if (ts) {
            SDL_Texture *tt = SDL_CreateTextureFromSurface(ren, ts);
            SDL_Rect dst = { panel.x + 20, y, ts->w, ts->h };
            SDL_RenderCopy(ren, tt, NULL, &dst);
            SDL_DestroyTexture(tt);
            y += ts->h + 20;
            SDL_FreeSurface(ts);
        }
    }

    for (int i = 0; i < count; i++) {
        SDL_Color col = (i == selected) ? yellow : white;
        SDL_Surface *s = TTF_RenderUTF8_Solid(font, items[i], col);
        if (!s) continue;
        SDL_Texture *t = SDL_CreateTextureFromSurface(ren, s);
        SDL_Rect dst = { panel.x + 40, y, s->w, s->h };
        SDL_RenderCopy(ren, t, NULL, &dst);
        SDL_DestroyTexture(t);
        SDL_FreeSurface(s);
        y += 30;
    }

    SDL_RenderPresent(ren);
}

static void load_settings(Settings *s)
{
    s->volume = 60;
    s->bg_index = 0;

    FILE *f = fopen(CFG_PATH, "r");
    if (!f) return;

    char line[128];
    while (fgets(line, sizeof(line), f)) {
        if (sscanf(line, "VOLUME=%d", &s->volume) == 1) continue;
        if (sscanf(line, "BG_INDEX=%d", &s->bg_index) == 1) continue;
    }
    fclose(f);

    if (s->volume < 0) s->volume = 0;
    if (s->volume > 100) s->volume = 100;
    if (s->bg_index < 0 || s->bg_index >= bg_key_count) s->bg_index = 0;
}

static void save_settings(const Settings *s)
{
    FILE *f = fopen(CFG_PATH, "w");
    if (!f) return;
    fprintf(f, "VOLUME=%d\n", s->volume);
    fprintf(f, "BG_INDEX=%d\n", s->bg_index);
    fclose(f);
}

static void run_settings_menu(SDL_Renderer *ren, TTF_Font *font)
{
    Settings s;
    load_settings(&s);

    int selected = 0; // 0 = volume, 1 = background, 2 = back
    int running = 1;
    Uint8 alpha = 180;

    while (running) {
        char items[3][256];
        snprintf(items[0], sizeof(items[0]), "Volume: %d", s.volume);
        snprintf(items[1], sizeof(items[1]), "Background: %s", bg_keys[s.bg_index]);
        snprintf(items[2], sizeof(items[2]), "Back");

        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            if (e.type == SDL_QUIT) return;
            if (e.type == SDL_KEYDOWN) {
                switch (e.key.keysym.sym) {
                case SDLK_ESCAPE:
                    running = 0;
                    break;
                case SDLK_UP:
                    if (selected > 0) selected--;
                    break;
                case SDLK_DOWN:
                    if (selected < 2) selected++;
                    break;
                case SDLK_LEFT:
                    if (selected == 0 && s.volume > 0) s.volume -= 5;
                    if (selected == 1 && s.bg_index > 0) s.bg_index--;
                    break;
                case SDLK_RIGHT:
                    if (selected == 0 && s.volume < 100) s.volume += 5;
                    if (selected == 1 && s.bg_index < bg_key_count - 1) s.bg_index++;
                    break;
                case SDLK_RETURN:
                case SDLK_SPACE:
                    if (selected == 2) running = 0;
                    break;
                default:
                    break;
                }
            }
        }

        draw_menu(ren, font, "Settings", items, 3, selected, alpha);
        SDL_Delay(16);
    }

    save_settings(&s);

    char cmd[128];
    snprintf(cmd, sizeof(cmd), "emu-bg %s &", bg_keys[s.bg_index]);
    system(cmd);
}

static void launch_retroarch(const char *core, const char *rom)
{
    const char *argv[] = {
        "/usr/bin/retroarch",
        "-L", core,
        rom,
        NULL
    };
    execvp(argv[0], (char * const *)argv);
    perror("execvp");
    _exit(1);
}

static int menu_loop(SDL_Renderer *ren, TTF_Font *font,
                     const char *title, char items[][256], int count)
{
    int selected = 0;
    int running = 1;
    Uint8 alpha = 0;

    SDL_GameControllerEventState(SDL_ENABLE);
    for (int i = 0; i < SDL_NumJoysticks(); i++) {
        if (SDL_IsGameController(i)) {
            SDL_GameControllerOpen(i);
        }
    }

    while (alpha < 180) {
        draw_menu(ren, font, title, items, count, selected, alpha);
        alpha += 10;
        SDL_Delay(16);
    }
    alpha = 180;

    while (running) {
        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            switch (e.type) {
            case SDL_QUIT:
                return -1;
            case SDL_KEYDOWN:
                if (e.key.keysym.sym == SDLK_ESCAPE) return -1;
                if (e.key.keysym.sym == SDLK_UP && selected > 0) selected--;
                else if (e.key.keysym.sym == SDLK_DOWN && selected < count - 1) selected++;
                else if (e.key.keysym.sym == SDLK_RETURN || e.key.keysym.sym == SDLK_SPACE)
                    return selected;
                break;
            case SDL_CONTROLLERBUTTONDOWN:
                if (e.cbutton.button == SDL_CONTROLLER_BUTTON_DPAD_UP && selected > 0)
                    selected--;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_DPAD_DOWN && selected < count - 1)
                    selected++;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_A)
                    return selected;
                else if (e.cbutton.button == SDL_CONTROLLER_BUTTON_B)
                    return -1;
                break;
            default:
                break;
            }
        }

        draw_menu(ren, font, title, items, count, selected, alpha);
        SDL_Delay(16);
    }
    return -1;
}

static int list_roms(const char *dir, char items[][256])
{
    DIR *d = opendir(dir);
    if (!d) return 0;
    struct dirent *ent;
    int n = 0;
    while ((ent = readdir(d)) && n < MAX_ITEMS) {
        if (ent->d_name[0] == '.') continue;
        snprintf(items[n], 256, "%s", ent->d_name);
        n++;
    }
    closedir(d);
    return n;
}

int main(int argc, char **argv)
{
    if (SDL_Init(SDL_INIT_VIDEO | SDL_INIT_GAMECONTROLLER) != 0) {
        fprintf(stderr, "SDL_Init: %s\n", SDL_GetError());
        return 1;
    }
    if (TTF_Init() != 0) {
        fprintf(stderr, "TTF_Init: %s\n", TTF_GetError());
        return 1;
    }

    SDL_Window *win = SDL_CreateWindow("emu-os",
                                       SDL_WINDOWPOS_UNDEFINED,
                                       SDL_WINDOWPOS_UNDEFINED,
                                       1280, 720,
                                       SDL_WINDOW_FULLSCREEN);
    if (!win) {
        fprintf(stderr, "SDL_CreateWindow: %s\n", SDL_GetError());
        return 1;
    }

    SDL_Renderer *ren = SDL_CreateRenderer(win, -1, SDL_RENDERER_ACCELERATED);
    if (!ren) {
        fprintf(stderr, "SDL_CreateRenderer: %s\n", SDL_GetError());
        return 1;
    }

    TTF_Font *font = TTF_OpenFont("/usr/share/fonts/ttf-dejavu/DejaVuSans.ttf", 24);
    if (!font) {
        fprintf(stderr, "TTF_OpenFont: %s\n", TTF_GetError());
        return 1;
    }

    int running = 1;
    while (running) {
        char sys_items[MAX_ITEMS][256];
        for (int i = 0; i < system_count; i++) {
            snprintf(sys_items[i], sizeof(sys_items[i]), "%s", systems[i].label);
        }
        snprintf(sys_items[system_count], sizeof(sys_items[system_count]), "Settings");
        int total = system_count + 1;

        int sys_idx = menu_loop(ren, font, "Select System", sys_items, total);
        if (sys_idx < 0) break;

        if (sys_idx == system_count) {
            run_settings_menu(ren, font);
            continue;
        }

        SystemEntry *sys = &systems[sys_idx];

        char cmd[128];
        snprintf(cmd, sizeof(cmd), "emu-bg %s &", sys->bg_key);
        system(cmd);

        char rom_items[MAX_ITEMS][256];
        int rom_count = list_roms(sys->rom_dir, rom_items);
        if (rom_count <= 0) continue;

        int rom_idx = menu_loop(ren, font, sys->label, rom_items, rom_count);
        if (rom_idx < 0) continue;

        char rom_path[512];
        snprintf(rom_path, sizeof(rom_path), "%s/%s", sys->rom_dir, rom_items[rom_idx]);

        launch_retroarch(sys->core_path, rom_path);
    }

    TTF_CloseFont(font);
    SDL_DestroyRenderer(ren);
    SDL_DestroyWindow(win);
    TTF_Quit();
    SDL_Quit();
    return 0;
}

#define FLAG_PATH "/config/.firstboot_done"

static void ensure_dir(const char *path) {
    mkdir(path, 0755);
}

static void prepare_rom_dirs(void) {
    ensure_dir("/mnt/roms");
    ensure_dir("/mnt/roms/snes");
    ensure_dir("/mnt/roms/nes");
    ensure_dir("/mnt/roms/genesis");
    ensure_dir("/mnt/roms/arcade");
    ensure_dir("/mnt/roms/ps1");
}

static void prepare_config_files(void) {
    ensure_dir("/config");
    FILE *f = fopen("/config/emu.conf", "w");
    if (f) {
        fprintf(f, "VOLUME=60\n");
        fprintf(f, "BG_INDEX=0\n");
        fclose(f);
    }
}

static void detect_controllers_and_audio(void) {
    SDL_InitSubSystem(SDL_INIT_GAMECONTROLLER | SDL_INIT_AUDIO);

    int pads = 0;
    for (int i = 0; i < SDL_NumJoysticks(); i++) {
        if (SDL_IsGameController(i)) pads++;
    }
    (void)pads;

    if (SDL_GetAudioDeviceName(0, 0)) {
        /* could store preferred device in a config later */
    }
}

int main(int argc, char **argv) {
    if (access(FLAG_PATH, F_OK) == 0) {
        return 0;
    }

    if (SDL_Init(SDL_INIT_VIDEO | SDL_INIT_GAMECONTROLLER | SDL_INIT_AUDIO) != 0) {
        return 1;
    }
    if (TTF_Init() != 0) {
        SDL_Quit();
        return 1;
    }

    SDL_Window *win = SDL_CreateWindow("emu-os setup",
                                       SDL_WINDOWPOS_UNDEFINED,
                                       SDL_WINDOWPOS_UNDEFINED,
                                       1280, 720,
                                       SDL_WINDOW_FULLSCREEN);
    if (!win) {
        TTF_Quit();
        SDL_Quit();
        return 1;
    }

    SDL_Renderer *ren = SDL_CreateRenderer(win, -1, SDL_RENDERER_ACCELERATED);
    if (!ren) {
        SDL_DestroyWindow(win);
        TTF_Quit();
        SDL_Quit();
        return 1;
    }

    TTF_Font *font = TTF_OpenFont("/usr/share/fonts/ttf-dejavu/DejaVuSans.ttf", 28);

    prepare_rom_dirs();
    prepare_config_files();
    detect_controllers_and_audio();

    int done = 0;
    while (!done) {
        SDL_Event e;
        while (SDL_PollEvent(&e)) {
            if (e.type == SDL_QUIT) done = 1;
            if (e.type == SDL_KEYDOWN) {
                if (e.key.keysym.sym == SDLK_RETURN || e.key.keysym.sym == SDLK_SPACE)
                    done = 1;
            }
            if (e.type == SDL_CONTROLLERBUTTONDOWN) {
                if (e.cbutton.button == SDL_CONTROLLER_BUTTON_A)
                    done = 1;
            }
        }

        SDL_SetRenderDrawColor(ren, 0, 0, 0, 255);
        SDL_RenderClear(ren);

        if (font) {
            SDL_Color white = {255, 255, 255, 255};
            SDL_Surface *s = TTF_RenderUTF8_Solid(font,
                "Welcome to emu-os! ROM storage and configs are prepared.",
                white);
            if (s) {
                SDL_Texture *t = SDL_CreateTextureFromSurface(ren, s);
                SDL_Rect dst = { 80, 300, s->w, s->h };
                SDL_RenderCopy(ren, t, NULL, &dst);
                SDL_DestroyTexture(t);
                SDL_FreeSurface(s);
            }
        }

        SDL_RenderPresent(ren);
        SDL_Delay(16);
    }

    FILE *flag = fopen(FLAG_PATH, "w");
    if (flag) fclose(flag);

    if (font) TTF_CloseFont(font);
    SDL_DestroyRenderer(ren);
    SDL_DestroyWindow(win);
    TTF_Quit();
    SDL_Quit();
    return 0;
}

cat > "$ROOT_DIR/install_emu_os_usb.sh" << 'EOF'
#!/usr/bin/env bash
set -e

if [ "$EUID" -ne 0 ]; then
    echo "Run as root"
    exit 1
fi

if [ $# -ne 3 ]; then
    echo "Usage: $0 <USB device> <bzImage> <rootfs.cpio.gz>"
    echo "Example: $0 /dev/sdb bzImage rootfs.cpio.gz"
    exit 1
fi

DEV="$1"
KERNEL="$2"
INITRD="$3"

echo "WARNING: This will erase ALL data on $DEV"
read -p "Continue? (yes/no): " yn
[ "$yn" = "yes" ] || exit 1

echo "Partitioning $DEV"
sgdisk --zap-all "$DEV"
sgdisk -n 1:0:+2G -t 1:8300 -c 1:"EMU-OS" "$DEV"
sgdisk -n 2:0:0   -t 2:0700 -c 2:"ROMDATA" "$DEV"

partprobe "$DEV"

P1="${DEV}1"
P2="${DEV}2"

echo "Formatting partitions"
mkfs.ext4 -L EMU-OS "$P1"
mkfs.exfat -n ROMDATA "$P2"

Will this build a 32bit or 64bit OS/system ?
Short answer: Your current configuration builds a 64‑bit OS.

Let me break it down cleanly so you see exactly why.

🧠 Why it’s 64‑bit
Inside your Buildroot defconfig, you have this line:

Code
BR2_x86_64=y
That single flag tells Buildroot:

Target architecture = x86_64

Kernel = 64‑bit

Toolchain = 64‑bit

Userland = 64‑bit

Everything else in the system inherits from that.

So the OS you’re building with the rebuild script is fully 64‑bit.

🧩 What if you wanted a 32‑bit build instead?
You would change:

Code
BR2_x86_64=y
to:

Code
BR2_i386=y
And optionally specify a CPU type:

Code
BR2_x86_i686=y
Then Buildroot would generate:

A 32‑bit kernel

A 32‑bit userland

32‑bit RetroArch + cores

32‑bit SDL2, mpv, ffmpeg, etc.

⚙️ Should you stay 64‑bit or switch?
Given your hardware:

Intel/AMD 1.5 GHz

4 GB RAM

256 GB HDD

A 64‑bit build is the right choice.
RetroArch cores, ffmpeg, mpv, SDL2, and GPU drivers all run better in 64‑bit mode on modern CPUs.

There’s no advantage to going 32‑bit unless:

You’re targeting Pentium 4 or older

You need to run 32‑bit‑only proprietary blobs

You want to reduce memory footprint to the absolute minimum

For your emulator appliance, 64‑bit is ideal.

No.  
Buildroot, the kernel toolchain, and the cross‑compiler do not run natively on Windows.

You must have:

GNU Make

GCC cross‑compiler

BusyBox tools

ELF toolchain

Linux filesystem semantics

Case‑sensitive paths

Symlinks

POSIX shell

Windows simply doesn’t provide these.

🧠 Can you build emu‑os using WSL2 on Windows?
YES — and this is the easiest Windows option.

WSL2 is a real Linux kernel running inside Windows.

You can install Ubuntu on WSL2 and run:

Code
sudo apt install build-essential git bc bison flex libssl-dev libncurses5-dev
make emu_x86_64_defconfig
make
This works perfectly and is the most convenient Windows‑based workflow.

🧠 Can you build emu‑os using VMware / VirtualBox?
YES — this is the most reliable option.

A Linux VM (Ubuntu, Debian, Arch, etc.) gives you:

Full Linux kernel

Full filesystem semantics

No weird Windows path issues

Perfect compatibility with Buildroot

This is the “classic” way to build embedded Linux systems.

🧠 Can you build emu‑os on a real Linux machine?
YES — this is the best option.

Native Linux builds are:

Fastest

Most stable

Easiest to debug

Zero virtualization overhead

If you have an old laptop or PC, this is ideal.

🧠 Which option should YOU use?
Given your setup and goals, here’s the ranking:

1. WSL2 (Windows Subsystem for Linux)
Fast, easy, no VM overhead.
Perfect for Buildroot.

2. VMware / VirtualBox Linux VM
Rock‑solid, isolated, predictable.

Best performance, but requires dedicated hardware.
